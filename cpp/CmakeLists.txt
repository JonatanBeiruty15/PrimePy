cmake_minimum_required(VERSION 3.15)
project(primepy_cpp VERSION 0.1.0 LANGUAGES CXX)

# ============================================
# Global C++ settings
# ============================================
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Debug)  # enable debug symbols by default

# ============================================
# External libraries
# ============================================
set(PYBIND11_DIR ${CMAKE_SOURCE_DIR}/external/pybind11)
set(EIGEN_DIR    ${CMAKE_SOURCE_DIR}/external/eigen)

add_subdirectory(${PYBIND11_DIR})
include_directories(${EIGEN_DIR})
include_directories(${CMAKE_SOURCE_DIR}/include)

# ============================================
# Core static library (C++ backend)
# ============================================
file(GLOB_RECURSE CORE_SOURCES src/*.cpp)
add_library(primepy_core STATIC ${CORE_SOURCES})
target_include_directories(primepy_core PUBLIC include)

# ============================================
# Utility Generators (precomputed data files)
# ============================================

# -- 1) Prime table generator --
add_executable(gen_primes_bin tools/number_theory/gen_primes_bin.cpp)

# Fixed size (your internal choice; adjust anytime)
set(PRIMEPY_FIXED_N_PRIMES 100000)

# Generated binary path (in build tree)
set(PRIMES_BIN ${CMAKE_BINARY_DIR}/primes_u32_le.bin)

# Rule to create the binary
add_custom_command(
    OUTPUT  ${PRIMES_BIN}
    COMMAND gen_primes_bin ${PRIMEPY_FIXED_N_PRIMES} ${PRIMES_BIN}
    DEPENDS gen_primes_bin
    COMMENT "Generating ${PRIMEPY_FIXED_N_PRIMES} primes → ${PRIMES_BIN}"
)

# Generate on normal builds
add_custom_target(generate_primes_bin ALL DEPENDS ${PRIMES_BIN})

# Ensure core builds after the data exists (safe even before bindings use it)
add_dependencies(primepy_core generate_primes_bin)

# Install the binary into the Python package data folder
install(FILES ${PRIMES_BIN}
        DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/../primepie/data)

# ============================================
# Build tests
# ============================================
file(GLOB_RECURSE TEST_SOURCES tests/*.cpp)
foreach(test_file ${TEST_SOURCES})
    get_filename_component(test_name ${test_file} NAME_WE)
    add_executable(${test_name} ${test_file})
    target_link_libraries(${test_name} primepy_core)
    message(STATUS "Adding test binary: ${test_name}")
endforeach()

enable_testing()
foreach(test_file ${TEST_SOURCES})
    get_filename_component(test_name ${test_file} NAME_WE)
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

add_custom_target(run_all_tests
    COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure > test_results.txt
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Running all tests and saving output to test_results.txt"
)

# ============================================
# Python bindings (_core)
# ============================================
file(GLOB_RECURSE BINDING_SOURCES bindings/*.cpp)
if (BINDING_SOURCES)
    pybind11_add_module(_core MODULE ${BINDING_SOURCES})
    target_link_libraries(_core PRIVATE primepy_core)

    # symbols are guaranteed to be in the .so
    target_sources(_core PRIVATE ${CORE_SOURCES})

    # Make sure the data binary exists first (convenient dependency)
    add_dependencies(_core generate_primes_bin)

    # Install the shared library into the Python package
    install(TARGETS _core
            LIBRARY DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/../primepie/core
            RUNTIME DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/../primepie/core)  # Windows
endif()
