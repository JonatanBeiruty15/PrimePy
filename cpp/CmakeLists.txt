cmake_minimum_required(VERSION 3.15)
project(primepy_cpp VERSION 0.1.0 LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Enable debug symbols by default
set(CMAKE_BUILD_TYPE Debug)

# External libraries
set(PYBIND11_DIR ${CMAKE_SOURCE_DIR}/external/pybind11)
set(EIGEN_DIR ${CMAKE_SOURCE_DIR}/external/eigen)

# Add pybind11
add_subdirectory(${PYBIND11_DIR})

# Include Eigen headers (header-only)
include_directories(${EIGEN_DIR})

# Include internal headers
include_directories(${CMAKE_SOURCE_DIR}/include)

#=============================
# Build core library (optional static lib for reuse)
#=============================
file(GLOB_RECURSE CORE_SOURCES src/*.cpp)
add_library(primepy_core STATIC ${CORE_SOURCES})
target_include_directories(primepy_core PUBLIC include)

#=============================
# Build tests
#=============================
file(GLOB_RECURSE TEST_SOURCES tests/*.cpp)
foreach(test_file ${TEST_SOURCES})
    get_filename_component(test_name ${test_file} NAME_WE)
    add_executable(${test_name} ${test_file})
    target_link_libraries(${test_name} primepy_core)
    message(STATUS "Adding test binary: ${test_name}")
endforeach()

# Register tests with CTest
enable_testing()
foreach(test_file ${TEST_SOURCES})
    get_filename_component(test_name ${test_file} NAME_WE)
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

#=============================
# Custom target to run all tests and save output
#=============================
add_custom_target(run_all_tests
    COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure > test_results.txt
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Running all tests and saving output to test_results.txt"
)

#=============================
# Build Python bindings
#=============================
file(GLOB_RECURSE BINDING_SOURCES bindings/*.cpp)
if (BINDING_SOURCES)
    pybind11_add_module(_core MODULE ${BINDING_SOURCES})  # Renamed from 'primepy' to '_core'
    target_link_libraries(_core PRIVATE primepy_core)

    # Install the shared library into the core folder of your Python package
    install(TARGETS _core
            LIBRARY DESTINATION ../primepy/core
            RUNTIME DESTINATION ../primepy/core)  # For Windows
endif()
install(TARGETS _core
        LIBRARY DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/../primepy/core
        RUNTIME DESTINATION ${CMAKE_CURRENT_SOURCE_DIR}/../primepy/core)
