\chapter{Algebra}

The algebra module is organized around two kinds of structures:
\begin{enumerate}
  \item \textbf{Groups}
  \item \textbf{Rings}
\end{enumerate}

Most algebra operations accept plain Python integers as inputs when this is
unambiguous, but they return \texttt{Element} objects. An element remembers the
group or ring it belongs to, prints in algebraic notation, and exposes its raw
Python value through \texttt{.value}.

For integer-valued elements, the C++ backend uses GMP's
\texttt{mpz\_class}. This means that algebraic elements are represented with
arbitrary-precision integers rather than fixed-width \texttt{int64\_t}
arithmetic.

\begin{pybox}
from primepie.algebra import AdditiveModGroup

G = AdditiveModGroup(7)
a = G(10)

print(a)        # 3 mod 7
print(a.value)  # 3
\end{pybox}

\section{Groups}

A group provides an identity element, inverses, a binary operation, and
exponentiation by repeated use of the group operation. In an additive group,
\texttt{power(a, n)} means adding \texttt{a} to itself \texttt{n} times.

\paragraph{Additive groups modulo n.}
\begin{pybox}
from primepie.algebra import AdditiveModGroup

G = AdditiveModGroup(7)

print(G.identity())       # 0 mod 7
print(G.inverse(3))       # 4 mod 7
print(G.operate(3, 6))    # 2 mod 7
print(G.power(2, 3))      # 6 mod 7
\end{pybox}

\paragraph{Batch exponentiation.}
\begin{pybox}
G = AdditiveModGroup(7)

same_exponent = G.power([1, 2, 3], 4)
different_exponents = G.power([1, 2, 3], [0, 1, 2])

print([x.value for x in same_exponent])       # [4, 1, 5]
print([x.value for x in different_exponents]) # [0, 2, 6]
\end{pybox}

\paragraph{Multiplicative groups modulo n.}
The multiplicative group modulo $n$ contains only the invertible residue classes,
that is, elements coprime to $n$.
\begin{pybox}
from primepie.algebra import MultiplicativeModGroup

U = MultiplicativeModGroup(7)

print(U.identity())       # 1 mod 7
print(U.inverse(3))       # 5 mod 7
print(U.operate(3, 5))    # 1 mod 7
print(U.power(3, 4))      # 4 mod 7
\end{pybox}

\paragraph{Direct sums of groups.}
\begin{pybox}
from primepie.algebra import AdditiveModGroup, DirectSumGroup

Z5 = AdditiveModGroup(5)
Z7 = AdditiveModGroup(7)
G = DirectSumGroup([Z5, Z7])

a = G((2, 3))
b = G((4, 6))

print(a)                # (2 mod 5, 3 mod 7)
print(G.operate(a, b))  # (1 mod 5, 2 mod 7)
print(G.power(a, 3))    # (1 mod 5, 2 mod 7)
\end{pybox}

\section{Rings}

A ring in PrimePie is also used as an additive group. This means
\texttt{power(a, n)} is additive power, while \texttt{mpower(a, n)} is
multiplicative power.

\paragraph{Rings modulo n.}
\begin{pybox}
from primepie.algebra import IntegersModRing

R = IntegersModRing(7)

print(R.zero())          # 0 mod 7
print(R.one())           # 1 mod 7
print(R.add(5, 6))       # 4 mod 7
print(R.mul(3, 5))       # 1 mod 7
print(R.is_equal(10, 3)) # True
\end{pybox}

\paragraph{Additive and multiplicative power.}
\begin{pybox}
R = IntegersModRing(7)

print(R.power(2, 3))     # 6 mod 7, additive: 2 + 2 + 2
print(R.mpower(3, 4))    # 4 mod 7, multiplicative: 3^4

out = R.mpower([1, 2, 3, 6], 3)
print([x.value for x in out])  # [1, 1, 6, 6]
\end{pybox}

\paragraph{The ring of integers.}
\begin{pybox}
from primepie.algebra import Integers

Z = Integers()

print(Z.add(5, 6))              # 11
print(Z.mul(5, 6))              # 30
print(Integers.gcd(84, 30))     # 6
print(Integers.is_prime(97))    # True
print(Integers.is_prime([2, 4, 5, 9]))
# [True, False, True, False]
\end{pybox}

\paragraph{Polynomial rings.}
\begin{pybox}
from primepie.algebra import IntegersModRing, PolynomialRing

R = IntegersModRing(5)
P = PolynomialRing(R)

f = P([1, 2])   # 1 + 2X
g = P([4, 1])   # 4 + X

print(f)            # (1 mod 5) + (2 mod 5) * X
print(P.add(f, g))  # (3 mod 5) * X
print(P.mul(f, g))  # (4 mod 5) + (4 mod 5) * X + (2 mod 5) * X^2
print(P.add(f, 3))  # (4 mod 5) + (2 mod 5) * X
\end{pybox}
